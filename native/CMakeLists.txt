# ub_native: WAMR and libriscv in one shared library for the Unity project.
cmake_minimum_required(VERSION 3.14)
project(ub_native C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(DEPS ${CMAKE_CURRENT_SOURCE_DIR}/../deps)
set(WAMR_ROOT_DIR ${DEPS}/wamr)
if(WIN32)
  set(WAMR_BUILD_PLATFORM "windows")
else()
  set(WAMR_BUILD_PLATFORM "darwin")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)")
  set(WAMR_BUILD_TARGET "AARCH64")
else()
  set(WAMR_BUILD_TARGET "X86_64")
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(vmlib STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(vmlib PUBLIC ${WAMR_ROOT_DIR}/core/iwasm/include)
target_link_libraries(vmlib PUBLIC m)

set(RISCV_32I OFF CACHE BOOL "" FORCE)
set(RISCV_64I ON CACHE BOOL "" FORCE)
set(RISCV_EXT_V OFF CACHE BOOL "" FORCE)
# Binary translation compiles C with the system compiler at run time; Windows has none.
if(WIN32)
  set(RISCV_BINARY_TRANSLATION OFF CACHE BOOL "" FORCE)
else()
  set(RISCV_BINARY_TRANSLATION ON CACHE BOOL "" FORCE)
endif()
set(RISCV_LIBTCC OFF CACHE BOOL "" FORCE)
set(RISCV_FLAT_RW_ARENA ON CACHE BOOL "" FORCE)
set(RISCV_THREADED ON CACHE BOOL "" FORCE)
add_subdirectory(${DEPS}/libriscv/lib libriscv)

add_library(ub_native SHARED ub_native.cpp)
target_include_directories(ub_native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ub_native PRIVATE vmlib riscv)
set_target_properties(ub_native PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(WIN32)
  set_target_properties(ub_native PROPERTIES PREFIX "")
endif()
